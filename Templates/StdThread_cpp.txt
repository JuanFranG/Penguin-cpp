#include <exception>
#include <iostream>
#include <sstream>
#include <string>
#include <thread>
#include <vector>

struct ThreadItem {
    std::string result;
    std::thread worker;
};

void ThreadFunction(std::string* result) {
    std::ostringstream text;
    text << "Hello World from thread ID " << std::this_thread::get_id();
    *result = text.str();
}

int main() {
    unsigned count = std::thread::hardware_concurrency();
    if (count == 0) count = 2;
    if (count > 4) count = 4;
    std::vector<ThreadItem> threads(count);

    try {
        for (unsigned i = 0; i < count; ++i)
            threads[i].worker = std::thread(ThreadFunction, &threads[i].result);
    } catch (const std::exception& error) {
        for (unsigned i = 0; i < count; ++i)
            if (threads[i].worker.joinable()) threads[i].worker.join();
        std::cerr << "Cannot start a thread: " << error.what() << '\n';
        return 1;
    }

    for (unsigned i = 0; i < count; ++i)
        threads[i].worker.join();
    for (unsigned i = 0; i < count; ++i)
        std::cout << threads[i].result << '\n';
    return 0;
}
