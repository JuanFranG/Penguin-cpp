/* C has no std::thread: use POSIX threads on macOS and Linux. */
#include <pthread.h>
#include <stdio.h>
#include <string.h>

struct ThreadItem {
    unsigned number;
    char result[80];
};

static void* ThreadFunction(void* argument) {
    struct ThreadItem* item = (struct ThreadItem*)argument;
    snprintf(item->result, sizeof item->result,
             "Hello World from worker %u", item->number);
    return NULL;
}

int main(void) {
    pthread_t workers[4];
    struct ThreadItem items[4];
    unsigned started = 0;
    int error = 0;

    for (unsigned i = 0; i < 4; ++i) {
        items[i].number = i + 1;
        error = pthread_create(&workers[i], NULL, ThreadFunction, &items[i]);
        if (error != 0) {
            fprintf(stderr, "Cannot start a thread: %s\n", strerror(error));
            break;
        }
        ++started;
    }
    for (unsigned i = 0; i < started; ++i) {
        int join_error = pthread_join(workers[i], NULL);
        if (join_error != 0) {
            fprintf(stderr, "Cannot join a thread: %s\n", strerror(join_error));
            return 1;
        }
        puts(items[i].result);
    }
    return error != 0;
}
