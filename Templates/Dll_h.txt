#ifndef _DLL_H_
#define _DLL_H_

/* A shared library (.so). Linux exports every function of a library unless
   it is built with -fvisibility=hidden; DLLIMPORT marks the ones to export
   in that case too. (On Windows this was __declspec(dllexport/dllimport).) */
#define DLLIMPORT __attribute__((visibility("default")))

DLLIMPORT void HelloWorld();

#endif
